cmake_minimum_required(VERSION 3.20)
project(MiniVectorDB LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(minivectordb_core src/types/value.cpp src/vector/exact.cpp src/storage/store.cpp)
# The portable reference must use distinct multiply/add too, including AArch64.
# Explicit SIMD screening intrinsics retain their intentional bounded FMAs.
set_source_files_properties(src/vector/exact.cpp PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")
target_include_directories(minivectordb_core PUBLIC include)
target_compile_options(minivectordb_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE minivectordb_core)
enable_testing()
foreach(suite types vector storage storage_failure)
  add_test(NAME core_${suite} COMMAND core_tests ${suite})
endforeach()
add_executable(namespace_tests tests/namespace_tests.cpp)
target_link_libraries(namespace_tests PRIVATE minivectordb_core)
add_test(NAME core_namespaces COMMAND namespace_tests)
target_sources(minivectordb_core PRIVATE src/storage/row_codec.cpp src/catalog/catalog.cpp src/sql/lexer.cpp src/sql/parser.cpp src/sql/binder.cpp src/sql/sorter.cpp src/sql/executor.cpp)
add_executable(sql_tests tests/sql_tests.cpp)
target_link_libraries(sql_tests PRIVATE minivectordb_core)
foreach(suite codec sql sort sql_crash sql_edges sql_failure sql_batch)
  add_test(NAME core_${suite} COMMAND sql_tests ${suite})
  set_tests_properties(core_${suite} PROPERTIES TIMEOUT 120)
endforeach()
set_tests_properties(core_storage core_storage_failure core_namespaces PROPERTIES TIMEOUT 180)
